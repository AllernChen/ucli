- [ ] **Step 3: 实现 `src/sessionBatch.js`**

```js
export function createBatchSelection() {
  let ids = new Set()
  return {
    toggle(id) { ids.has(id) ? ids.delete(id) : ids.add(id) },
    setAll(allIds) { ids = new Set(allIds) },
    isAllSelected(allIds) { return allIds.length > 0 && allIds.every(id => ids.has(id)) },
    selected() { return ids },
    mode() { return ids.size > 0 },
    clear() { ids = new Set() }
  }
}
```
